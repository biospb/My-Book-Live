echo ==== MBL multiboot: bootsel=${bootsel} boot_count=${boot_count} ====

setenv os_0 'setenv osdir debian; setenv rootargs root=/dev/sda4 rw rootfstype=ext4 rootflags=data=ordered ipv6.disable=1 libata.force=noncq rng_core.default_quality=700'
setenv os_1 'setenv osdir owrt; setenv rootargs root=/dev/sda2 rw rootfstype=squashfs,ext4'
setenv os_2 'setenv osdir debian-test; setenv rootargs root=/dev/sda4 rw rootfstype=ext4 rootflags=data=ordered ipv6.disable=1 libata.force=noncq rng_core.default_quality=700'
setenv flip_0 'setenv bootsel 1'
setenv flip_1 'setenv bootsel 0'
setenv flip_2 'setenv bootsel 0'

if itest ${boot_count} == 2; then setenv boot_count 3; fi
if itest ${boot_count} == 1; then setenv boot_count 2; fi
if itest ${boot_count} == 0; then setenv boot_count 1; fi
if itest ${boot_count} == 3; then echo "!!! two failed boots in a row - switching system"; run flip_${bootsel}; setenv boot_count 1; fi
saveenv

setenv ipaddr 192.168.7.4
setenv ncip 192.168.7.36
setenv ncmac 74:56:3c:30:c9:aa
setenv stdout nc
setenv stderr nc

run os_${bootsel}
echo ==== booting ${osdir} (bootsel=${bootsel}, attempt ${boot_count}) ====
setenv bootargs ${rootargs} panic=10 netconsole=6663@${ipaddr}/eth0,6664@${ncip}/${ncmac}
run addtty
setenv load_os 'sata init; ext2load sata 1:1 ${kernel_addr_r} /boot/${osdir}/uImage; ext2load sata 1:1 ${fdt_addr_r} /boot/${osdir}/apollo3g.dtb'
if run load_os; then echo "==== loaded /boot/${osdir}"; else echo "!!! cannot load /boot/${osdir} - switching system"; run flip_${bootsel}; setenv boot_count 1; saveenv; reset; fi
bootm ${kernel_addr_r} - ${fdt_addr_r}
echo "!!! bootm failed"
reset
